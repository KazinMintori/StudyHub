$$
\begin{aligned}
I_1 &= \frac{2\mathcal E_1+\mathcal E_2}{3R}, \\
I_2 &= -\frac{\mathcal E_1+2\mathcal E_2}{3R}, \\
I_3 &= \frac{\mathcal E_1-\mathcal E_2}{3R}.
\end{aligned}
$$
