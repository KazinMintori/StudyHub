$$
E=\frac{\alpha}{2\varepsilon_0}

$$

$$
E_{\mathrm{trong}}=\frac{kQ}{R^2}(4u-3u^2).
$$

$$
T=2\pi R_o\sqrt{\frac{2\pi\varepsilon_0M}{Q\lambda}}.
$$

$$
\begin{aligned}
\Phi_{\mathrm{side}} &= Q\eta/\varepsilon_0, \\
\Phi_{\mathrm{base}} &= Q(1-\eta)/(2\varepsilon_0).
\end{aligned}
$$
